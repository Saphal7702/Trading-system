/***********
    FILTERS:
    Layer1: Eligibility — pure structural filters - (universe_builder filter)
      - US-domiciled common stock or REIT
      - Primary listing on NYSE/Nasdaq (not NasdaqCM penny tier, not OTC)
      - Market cap ≥ $X billion (S&P uses $8B; you might use $2-5B)
      - ADV20 ≥ $25M (real liquidity for slippage assumptions)
      - Price ≥ $10 (avoids penny-stock mechanics: tick noise, exchange rules, halt risk)
      - Public history ≥ N years (e.g., 3 years post-IPO so backtest periods are valid)
      - Not currently halted, delisted, in bankruptcy

    New Universe      : trading build-custom-universe --universe-name coreAlpha --source alpaca --use-backtest-db --top 0 --report
    Existing Universe : trading audit-universe --universe sp500 --dry-run
                        trading audit-universe --universe sp500
                        trading audit-universe --universe sp500 --remove-from-universe --report

    Layer 2: Fitness — statistical, evidence-based - (Back test filter)
      - Trend (have meaningful uptrends to ride)
      - Mean-revert within the trend (RSI2 oversold then bounces, instead of crashing through)
      - Don't gap violently on news (so your stops are honored)
      - Have reasonable volatility (enough movement to generate trades, not so much that ATR exits whip)
    
    New Universe      :  trading trim-universe --backtest-run-id 1 --from-universe coreAlpha --to-universe coreAlpha_r1 --min-entries 3 --max-wr-chronic 35 --min-net-pnl -10 --min-entries-consistent 5 --report 
    Existing Universe :  trading trim-universe --backtest-run-id <N> --dry-run
                         trading trim-universe --backtest-run-id <N>
                         trading trim-universe --backtest-run-id <N> --remove-from-universe --report
************/

/********************************************************************************* Back Testing ****************************************************************************************************************/

Step 1 — Fetch historical bars (only needed once per date range):                                                                                                                                                                                       
  trading backtest-fetch-bars --start 2020-01-01 --end 2024-12-31 (Only SP500)  
  trading backtest-fetch-bars --start 2020-01-01 --end 2024-12-31 --source-alpaca (All tradeable Symbols)                                                                                                                                                                       
                                                                                                                                                                                                                                                        
Step 2 — Run the backtest:    
  # Main command:
  trading backtest --simulate-live --universe coreAlpha --start 2020-01-01 --end 2025-12-31 --initial-capital 1000 --max-positions 10 --per-position 100 --compound

  # Default: both strategies, regime gates which signals fire each day                                                                                                                                                                                    
  trading backtest --start 2020-01-01 --end 2024-12-31                                                                                                                                                                                          
                                                                                                                                                                                                                                                          
  # Restrict to a single strategy                                                                                                                                                                                                                         
  trading backtest --start 2020-01-01 --end 2024-12-31 --strategy sma                                                                                                                                                                           
  trading backtest --start 2020-01-01 --end 2024-12-31 --strategy mrit                                                                                                                                                                          
                                                                                                                                                                                                                                                          
  # Custom capital / position sizing                                                                                                                                                                                                                      
  trading backtest --start 2020-01-01 --end 2024-12-31 --initial-capital 50000 --max-positions 10 --per-position 5000                                                                                                                                                                                        
                                                                                                                                                                                                                                                          
  # Tighten exit rules                                                                                                                                                                                                                                    
  trading backtest --start 2020-01-01 --end 2024-12-31 --early-fail-days 5 --early-fail-max-ret -3.0 --cooldown-days 14 

  trading backtest --start 2024-01-01 --end 2024-12-31 --initial-capital 500 --max-positions 5 --per-position 100 --early-fail-days 999 --cooldown-days 30
                                                                                                                                                                                                                                                          
All options:                                              
  --strategy {sma,mrit}     Which strategy to backtest (required)                                                                                                                                                                                         
  --start YYYY-MM-DD         Start date (required)                                                                                                                                                                                                        
  --end YYYY-MM-DD           End date (required)  
  --universe sp500           Universe of symbols (default: sp500)                                                                                                                                                                                         
  --initial-capital 100000   Starting cash (default: $100,000)                                                                                                                                                                                            
  --max-positions 20         Max open positions at once (default: 20)
  --per-position 5000        Notional per trade (default: capital / max-positions)                                                                                                                                                                        
  --fast 20                  SMA fast period for SMA strategy (default: 20)       
  --slow 50                  SMA slow period for SMA strategy (default: 50)                                                                                                                                                                               
  --db backtest.sqlite       Where to save results (default: backtest.sqlite)                                                                                                                                                                                                  
                                                                                                                                                                                                                                                          
Results are saved to backtest.sqlite and a performance report is printed to the terminal showing total return, Sharpe ratio, max drawdown, win rate, and the top trades by P&L.

SOURCE_DB="/home/saphal7702/Trading/Trading-paper/TradingData/trading.sqlite"
TARGET_DB="/home/saphal7702/Trading/Trading-live/TradingData/trading.sqlite"

# backup target first
cp "$TARGET_DB" "${TARGET_DB}.bak_$(date +%Y%m%d_%H%M%S)"

# export both tables from source
sqlite3 "$SOURCE_DB" ".dump universe_membership symbol_exclusions" > /tmp/universe_tables.sql

# drop both tables from target
sqlite3 "$TARGET_DB" "
DROP TABLE IF EXISTS universe_membership;
DROP TABLE IF EXISTS symbol_exclusions;
"

# import both tables into target
sqlite3 "$TARGET_DB" < /tmp/universe_tables.sql

# verify row counts
sqlite3 "$TARGET_DB" "
SELECT 'universe_membership' AS table_name, COUNT(*) FROM universe_membership
UNION ALL
SELECT 'symbol_exclusions' AS table_name, COUNT(*) FROM symbol_exclusions;
"

# verify strategy_scope backfill (should show 'all'=data_anomaly rows, 'mrit'=chronic_loser+others)
sqlite3 "$TARGET_DB" "
SELECT reason_code, strategy_scope, COUNT(*) AS cnt
FROM symbol_exclusions
WHERE reinstated_at IS NULL
GROUP BY reason_code, strategy_scope
ORDER BY strategy_scope, reason_code;
"

# confirm no data_anomaly rows ended up with non-'all' scope (should return 0 rows)
sqlite3 "$TARGET_DB" "
SELECT symbol, reason_code, strategy_scope
FROM symbol_exclusions
WHERE reason_code = 'data_anomaly' AND strategy_scope != 'all';
"

/***********************************  Git Commands (This Project):  **********************************************************/
git branch                  -> Show Branches
git checkout branchName     -> Switch to the Branch

git add .                   -> Puts a file from the Working Directory into the Staging Area of the Branch.
git commit -m "message"     -> Permanently records the staged changes as a new commit in the Branch.
git push                    - > Push the branch to the remote

git pull
git merge --no-ff paper
git push

/***********************************  Git Other Commands:  **********************************************************/
#All Git Commands:
git init	                -> Creates a new Git repository in the current directory.
git status	                -> Shows the state of your working directory and staging area (which files are modified, staged, or untracked).
git log	                    -> Shows the commit history
git checkout -b branchName  -> Create a new Branch
git fetch
git branch -a
git checkout main           -> Switch to Main
git pull origin main        -> Pull from Main
git merge branchName        -> Merge the Branch
git push origin main        -> Push to the Main
git branch -d branchName    -> Delete the branch
